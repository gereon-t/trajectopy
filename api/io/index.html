    def read_trajectory(
    filename: str | os.PathLike,
    *,
    format: "str | TrajectoryFormat" = TrajectoryFormat.AUTO,  # noqa: A002
) -> Trajectory:
    """Reads a trajectory file in any supported format.

    Args:
        filename (str | os.PathLike): Path to the file.
        format (str | TrajectoryFormat, optional): One of ``auto``, ``trajectopy``, ``tum``, ``kitti`` and
            ``euroc``. With ``auto`` (default), the format is recognized by [`detect_format`][trajectopy.io.detect_format].

    Returns:
        Trajectory: The loaded trajectory.

    See Also:
        [`trajectopy.Trajectory.from_file`][trajectopy.Trajectory.from_file]: the same as a class method.
        [`trajectopy.io.write_trajectory`][trajectopy.io.write_trajectory]: writes trajectories.

    Examples:
        >>> from trajectopy import io
        >>> trajectory = io.read_trajectory("./example_data/KITTI_gt.traj")
        >>> io.write_trajectory(trajectory, "poses.txt", format="kitti")
        >>> io.detect_format("poses.txt")
        <TrajectoryFormat.KITTI: 'kitti'>
        >>> len(io.read_trajectory("poses.txt"))
        4541
    """
    from trajectopy.io import trajectory_io

    filename = os.fspath(filename)
    file_format = TrajectoryFormat(format)
    if file_format == TrajectoryFormat.AUTO:
        file_format, has_fields = _scan_format(filename)
        if file_format != TrajectoryFormat.TRAJECTOPY:
            logger.info("Recognized %s as a %s file.", filename, _DISPLAY_NAMES[file_format])
        elif not has_fields:
            logger.info(
                "%s has no #fields header: its columns are read as t,px,py,pz,qx,qy,qz,qw (TUM format).", filename
            )

    if file_format == TrajectoryFormat.TRAJECTOPY:
        return trajectory_io.read_trajectory(filename, default_name=_name(filename))
    readers: dict[TrajectoryFormat, Callable[[str], Trajectory]] = {
        TrajectoryFormat.TUM: read_tum,
        TrajectoryFormat.KITTI: read_kitti,
        TrajectoryFormat.EUROC: read_euroc,
    }
    return readers[file_format](filename)

    def write_trajectory(
    trajectory: Trajectory,
    filename: str | os.PathLike,
    *,
    format: "str | TrajectoryFormat" = TrajectoryFormat.AUTO,  # noqa: A002
    mode: str = "w",
) -> None:
    """Writes a trajectory in one of the supported formats.

    Args:
        trajectory (Trajectory): Trajectory to write.
        filename (str | os.PathLike): Output path.
        format (str | TrajectoryFormat, optional): One of ``auto`` (default), ``trajectopy``, ``tum``,
            ``kitti`` and ``euroc``. ``auto`` picks the format from the file extension (``.tum``, ``.kitti``)
            and writes trajectopy files otherwise.
        mode (str, optional): ``"w"`` (default) overwrites the file, ``"a"`` appends to it.

    See Also:
        [`trajectopy.io.read_trajectory`][trajectopy.io.read_trajectory]: reads all formats.

    Examples:
        >>> from trajectopy import io
        >>> trajectory = io.read_trajectory("./example_data/KITTI_gt.traj")
        >>> io.write_trajectory(trajectory, "poses.tum")  # format from the extension
        >>> io.write_trajectory(trajectory, "poses.csv", format="euroc")
    """
    from trajectopy.io import trajectory_io

    filename = os.fspath(filename)
    file_format = TrajectoryFormat(format)
    if file_format == TrajectoryFormat.AUTO:
        suffixes = {".tum": TrajectoryFormat.TUM, ".kitti": TrajectoryFormat.KITTI}
        file_format = suffixes.get(Path(filename).suffix.lower(), TrajectoryFormat.TRAJECTOPY)

    if file_format == TrajectoryFormat.TRAJECTOPY:
        trajectory_io.write_trajectory(trajectory, filename, mode=mode)
        return

    writers = {
        TrajectoryFormat.TUM: to_tum_string,
        TrajectoryFormat.KITTI: to_kitti_string,
        TrajectoryFormat.EUROC: to_euroc_string,
    }
    with open(filename, mode, newline="\n", encoding="utf-8") as file:
        file.write(writers[file_format](trajectory))

    TrajectoryFormat ¶

    Bases: OptionEnum

    File formats of trajectories (see trajectopy.io.read_trajectory).

    Each option can also be given as its value, e.g. format="tum" (case-insensitive).

    Attributes:

    • AUTO –

      "auto": recognized from the content when reading, from the extension when writing.

    • TRAJECTOPY –

      "trajectopy": ASCII file with a header that describes the columns.

    • TUM –

      "tum": timestamp tx ty tz qx qy qz qw per line.

    • KITTI –

      "kitti": one row-major 3×4 pose matrix per line, without timestamps.

    • EUROC –

      "euroc": CSV with timestamps in nanoseconds and quaternions in the order w, x, y, z.

    Examples:

    >>> from trajectopy import io
>>> io.TrajectoryFormat("TUM")
<TrajectoryFormat.TUM: 'tum'>

    def read_tum(filename: str | os.PathLike, *, name: str = "") -> Trajectory:
    """Reads a trajectory in the TUM format (``timestamp tx ty tz qx qy qz qw``).

    Args:
        filename (str): Path to the file.
        name (str, optional): Name of the trajectory. Defaults to the file name.

    Returns:
        Trajectory: The trajectory, in a local coordinate system (EPSG 0).

    See Also:
        [`trajectopy.io.read_trajectory`][trajectopy.io.read_trajectory]: recognizes the format.

    Examples:
        >>> from trajectopy import io
        >>> io.write_trajectory(io.read_trajectory("./example_data/KITTI_gt.traj"), "poses.tum")
        >>> len(io.read_tum("poses.tum", name="KITTI"))
        4541
    """
    filename = os.fspath(filename)
    data = _numeric_table(filename, delimiter=None)
    if data.ndim != 2 or data.shape[1] < 8:
        raise TrajectopyIOError(
            f"A TUM file has 8 columns (timestamp tx ty tz qx qy qz qw), {filename} has {data.shape[-1]}."
        )
    return Trajectory(
        positions=Positions(xyz=data[:, 1:4]),
        rotations=Rotation.from_quat(data[:, 4:8]),
        timestamps=data[:, 0],
        name=name or _name(filename),
    )

    def read_kitti(
    filename: str | os.PathLike, *, timestamps: np.ndarray | str | None = None, name: str = ""
) -> Trajectory:
    """Reads a trajectory in the KITTI odometry format (one row-major 3x4 pose matrix per line).

    Args:
        filename (str): Path to the file.
        timestamps (np.ndarray | str | None, optional): Timestamps of the poses, or the path to a file with one
            timestamp per line (KITTI's ``times.txt``). Without them, the poses are numbered 0, 1, 2, ...
        name (str, optional): Name of the trajectory. Defaults to the file name.

    Returns:
        Trajectory: The trajectory, in a local coordinate system (EPSG 0).

    See Also:
        [`trajectopy.io.read_trajectory`][trajectopy.io.read_trajectory]: recognizes the format.

    Examples:
        >>> from trajectopy import io
        >>> trajectory = io.read_trajectory("./example_data/KITTI_gt.traj")
        >>> io.write_trajectory(trajectory, "poses.txt", format="kitti")
        >>> io.read_kitti("poses.txt").timestamps[:3]  # numbered without timestamps
        array([0., 1., 2.])
    """
    filename = os.fspath(filename)
    data = _numeric_table(filename, delimiter=None)
    if data.ndim != 2 or data.shape[1] != _KITTI_COLUMNS:
        raise TrajectopyIOError(
            f"A KITTI file has 12 values per line (3x4 pose matrix), {filename} has {data.shape[-1]}."
        )

    matrices = data.reshape(-1, 3, 4)
    if isinstance(timestamps, str | Path):
        timestamps = np.loadtxt(timestamps, comments="#", ndmin=1)
    if timestamps is not None and len(timestamps) != len(matrices):
        raise ValueError(f"{len(timestamps)} timestamps were given for {len(matrices)} poses.")

    return Trajectory(
        positions=Positions(xyz=matrices[:, :, 3].copy()),
        rotations=Rotation.from_matrix(matrices[:, :, :3]),
        timestamps=None if timestamps is None else np.asarray(timestamps, dtype=float),
        name=name or _name(filename),
    )

    def read_euroc(filename: str | os.PathLike, *, name: str = "") -> Trajectory:
    """Reads a trajectory in the EuRoC ground truth format.

    The columns are ``timestamp [ns], px, py, pz, qw, qx, qy, qz``, optionally followed by velocities
    ``vx, vy, vz`` and sensor biases (ignored).

    Args:
        filename (str): Path to the file.
        name (str, optional): Name of the trajectory. Defaults to the file name.

    Returns:
        Trajectory: The trajectory, in a local coordinate system (EPSG 0), with timestamps in seconds.

    See Also:
        [`trajectopy.io.read_trajectory`][trajectopy.io.read_trajectory]: recognizes the format.

    Examples:
        >>> from trajectopy import io
        >>> io.write_trajectory(io.read_trajectory("./example_data/KITTI_gt.traj"), "poses.csv", format="euroc")
        >>> len(io.read_euroc("poses.csv"))
        4541
    """
    filename = os.fspath(filename)
    data = _numeric_table(filename, delimiter=",")
    if data.ndim != 2 or data.shape[1] < 8:
        raise TrajectopyIOError(
            f"A EuRoC file has at least 8 columns (timestamp, position, quaternion), {filename} has {data.shape[-1]}."
        )
    return Trajectory(
        positions=Positions(xyz=data[:, 1:4]),
        rotations=Rotation.from_quat(data[:, [5, 6, 7, 4]]),
        timestamps=data[:, 0] * 1e-9,
        velocity_xyz=data[:, 8:11] if data.shape[1] >= 11 else None,
        name=name or _name(filename),
    )

    def trajectory_to_string(trajectory: Trajectory) -> str:
    """Returns a trajectory in the trajectopy ASCII format, as [`write_trajectory`][trajectopy.io.write_trajectory] writes it.

    The header contains the EPSG code, the name, the navigation frame (always ENU), the sorting and the
    fields ``t,l,px,py,pz,qx,qy,qz,qw,vx,vy,vz`` (without the quaternion for trajectories without

    Args:
        trajectory (Trajectory): Trajectory to convert.

    Returns:
        str: Content of a trajectory file.

    Examples:
        >>> from trajectopy import io
        >>> trajectory = io.read_trajectory("./example_data/KITTI_gt.traj")

        f"#epsg {trajectory.positions.epsg}",
        f"#name {trajectory.name}",
        "#nframe enu",

        f"#fields {fields}",

            trajectory.timestamps, trajectory.path_lengths, trajectory.positions.xyz, trajectory.velocity_xyz
        ]
    else:

            trajectory.timestamps,

            trajectory.velocity_xyz,

    def trajectories_from_rosbag(filename: str | os.PathLike, *, topics: Iterable[str] | None = None) -> list[Trajectory]:
    """Creates trajectories from a ROS1 or ROS2 bag.

    Args:
        filename (str): Path to the bag: a ``.bag`` (ROS1), ``.mcap`` or ``.db3`` file, or a ROS2 bag directory.
        topics (Iterable[str] | None, optional): Topics to read. Defaults to all topics with a supported
            message type.

    Returns:
        list[Trajectory]: One trajectory per topic (and per frame pair of transform topics).

    See Also:
        [`trajectopy.io.list_topics`][trajectopy.io.list_topics]: lists the topics of a bag.

    Examples:
        >>> from trajectopy import io
        >>> trajectories = io.trajectories_from_rosbag("recording.mcap", topics=["/odom"])
        >>> [trajectory.name for trajectory in trajectories]
        ['/odom']
    """
    filename = os.fspath(filename)
    data = read_ros_bag(filename, topics=topics)

    trajectories = []
    for traj_key, traj_data in data.items():
        if not traj_data.get("x"):
            continue
        epsg = int(traj_data["epsg"][0]) if "epsg" in traj_data else 0
        pos = Positions(xyz=np.c_[traj_data["x"], traj_data["y"], traj_data["z"]], epsg=epsg)

        if any(len(traj_data.get(key, [])) != len(pos) for key in ["qx", "qy", "qz", "qw"]):
            rot = None
            logger.info("No rotation data found for trajectory %s", traj_key)
        else:
            rot = Rotation.from_quat(quat=np.c_[traj_data["qx"], traj_data["qy"], traj_data["qz"], traj_data["qw"]])

        position_std = None
        if all(len(traj_data.get(key, [])) == len(pos) for key in ["sx", "sy", "sz"]):
            position_std = np.c_[traj_data["sx"], traj_data["sy"], traj_data["sz"]]

        tstamps = np.array(traj_data.get("tstamps", np.arange(len(pos))), dtype=float)

        trajectories.append(
            Trajectory(name=traj_key, positions=pos, rotations=rot, timestamps=tstamps, position_std=position_std)
        )
        logger.info("Found trajectory: %s", traj_key)

    logger.info("Created %d trajectories from ROS bag %s", len(trajectories), filename)
    return trajectories

    def read_ate_result(filename: str | os.PathLike) -> ATEResult:
    """Reads an ATE result written by [`write_ate_result`][trajectopy.io.write_ate_result].

    Files written by trajectopy versions with an earlier sign convention of the deviations are
    converted to the current one (test minus reference).

    Args:
        filename (str | os.PathLike): Path of the file.

    Returns:
        ATEResult: The result. Its ``alignment`` is None, since result files do not store it.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> ate_result = tpy.ate(estimate, reference)
        >>> from trajectopy import io
        >>> io.write_ate_result(ate_result, "ate.result")
        >>> io.read_ate_result("ate.result").alignment is None
        True
    """
    filename = os.fspath(filename)
    header_data = HeaderData.from_file(filename)
    deviation_data = pd.read_csv(filename, comment="#")
    sorting = Sorting(header_data.sorting)

    # the rows in the order the trajectory will have, so that the deviations stay with their poses: one pose
    # per timestamp and, if sorted by path length, ties in chronological order
    tstamps = deviation_data["time"].to_numpy(dtype=float)
    chronological = np.argsort(tstamps, kind="stable")
    order = chronological[np.diff(tstamps[chronological], prepend=np.nan) != 0]
    if sorting == Sorting.PATH_LENGTH:
        path_lengths = deviation_data["arc_lengths"].to_numpy(dtype=float)
        order = order[np.argsort(path_lengths[order], kind="stable")]
    deviation_data = deviation_data.iloc[order]

    tstamps = deviation_data["time"].to_numpy(dtype=float)
    path_lengths = deviation_data["arc_lengths"].to_numpy(dtype=float)
    positions = Positions(deviation_data[["pos_x", "pos_y", "pos_z"]].to_numpy(dtype=float), epsg=header_data.epsg)

    rot_columns = ["rot_x", "rot_y", "rot_z", "rot_w"]
    if all(column in deviation_data.columns for column in rot_columns):
        rotations = Rotation.from_quat(deviation_data[rot_columns].to_numpy(dtype=float))
    else:
        rotations = None

    pos_dev = deviation_data[["pos_dev_x", "pos_dev_y", "pos_dev_z"]].to_numpy(dtype=float)
    directed_pos_dev = deviation_data[["pos_dev_along", "pos_dev_cross_h", "pos_dev_cross_v"]].to_numpy(dtype=float)

    rot_dev_columns = ["rot_dev_x", "rot_dev_y", "rot_dev_z", "rot_dev_w"]
    if all(column in deviation_data.columns for column in rot_dev_columns):
        rot_dev = Rotation.from_quat(deviation_data[rot_dev_columns].to_numpy(dtype=float))
    else:
        rot_dev = None

    trajectory = Trajectory(
        name=header_data.name,
        positions=positions,
        rotations=rotations,
        timestamps=tstamps,
        path_lengths=path_lengths,
        sorting=sorting,
    )
    pos_dev, directed_pos_dev, rot_dev = convert_ate_deviations(
        header_data.deviations, pos_dev, directed_pos_dev, rot_dev
    )
    if directed_by := str(header_data.data.get("directed_by", "")):
        rotations_used = directed_by == "rotations"
    else:
        # earlier versions did not write it: rotations were used if either trajectory had some
        rotations_used = rot_dev is not None or rotations is not None
    abs_dev = AbsoluteTrajectoryDeviations(
        pos_dev=pos_dev, directed_pos_dev=directed_pos_dev, rot_dev=rot_dev, rotations_used=rotations_used
    )
    return ATEResult(trajectory=trajectory, abs_dev=abs_dev)

        rot_dev=rot_dev,

        ate_result (ATEResult): Result to export. Its trajectory needs a known EPSG code.
        filename (str): The output filename (e.g., "ate.kml").
        field (str, optional): Deviation field that colors the line: "position", "x", "y", "z", "along_track",
            "horizontal_cross_track", "vertical_cross_track" and, with orientations, "rotation", "roll", "pitch"
            or "yaw". Defaults to "position".
        colormap (str, optional): Color scale, see [`write_kml`][trajectopy.io.write_kml]. Defaults to "rdylbu".
        max_std (float, optional): The color scale is clipped to mean ± ``max_std`` standard deviations
            (0 = minimum to maximum). Ignored if ``value_range`` is given. Defaults to 3.0, like the plots.
        value_range (tuple[float, float], optional): Fixed (lower, upper) limits of the color scale.
            Defaults to none.
        color_classes (int, optional): Number of colors of the scale. Defaults to 11.
        heights (str, optional): Vertical placement, see [`write_kml`][trajectopy.io.write_kml]. Defaults to "ground".
        height_offset (float, optional): Meters added to the heights, see [`write_kml`][trajectopy.io.write_kml]. Defaults to 0.
        extrude (bool, optional): Connect the line to the ground with a vertical curtain. Defaults to False.
        line_width (float, optional): Line width in pixels. Defaults to 4.
        precision (float, optional): Coordinate precision in degrees. Defaults to 1e-6 (about 0.1 m).

    Raises:
        ValueError: If the trajectory has no known EPSG code, the field is not available, an option is invalid,
            or the geoid model is not available.

    See Also:
        [`trajectopy.ATEResult.to_kml`][trajectopy.ATEResult.to_kml]: the same as a method.

    Examples:
        >>> import numpy as np
        >>> import trajectopy as tpy
        >>> from trajectopy import io
        >>> t = np.arange(0.0, 100.0)
        >>> xyz = np.column_stack([364000.0 + t, 5621000.0 + np.sin(t / 10), np.full_like(t, 60.0)])
        >>> reference = tpy.Trajectory(tpy.Positions(xyz, epsg=25832), timestamps=t)
        >>> estimate = tpy.Trajectory(tpy.Positions(xyz + 0.1, epsg=25832), timestamps=t)
        >>> ate_result = tpy.ate(estimate, reference, align=False)
        >>> io.write_ate_kml(ate_result, "ate.kml", field="along_track")
    """
    filename = os.fspath(filename)
    selected = _field(ate_kml_fields(ate_result), field)
    scale = _ColorScale(colormap, max_std, value_range, color_classes)
    placement = _placement(ate_result.trajectory, heights, height_offset)
    values = np.asarray(selected.values(ate_result), dtype=float)[placement.order]
    root, document = _document(ate_result.name)
    _add_colored_line(document, ate_result.name, selected, values, placement, scale, extrude, line_width, precision)
    _write(root, filename)
