    ATEResult(
    trajectory: Trajectory,
    abs_dev: AbsoluteTrajectoryDeviations,
    *,
    name: str = "",
    alignment: AlignmentResult | None = None,
)

    This class represents a set of absolute trajectory deviations

    Absolute trajectory deviations describe absolute pose deviations between two trajectories. The deviations are calculated by comparing pairs of positions and orientations in the test and reference trajectory.

    Attributes:

    • trajectory (Trajectory) –

      The trajectory the deviations are belonging to

    • abs_dev (AbsoluteTrajectoryDeviations) –

      The absolute trajectory deviations

    • name (str) –

      The name of the result. Defaults to the name of the trajectory.

    • alignment (AlignmentResult | None) –

      The alignment that was applied to the trajectory before the deviations were computed, or None if unknown (e.g. for results read from a file) or if none was applied. Defaults to None.

    See Also

    trajectopy.ate: computes the result.

    Examples:

    >>> import trajectopy as tpy
>>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
>>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
>>> ate_result = tpy.ate(estimate, reference)
>>> ate_result
<ATEResult 'KITTI ORB vs. KITTI Ground Truth': 4541 poses, position ATE 1.1533 m, rotation ATE 0.6204 deg>
>>> len(ate_result.pos_dev_along) == len(ate_result)
True

    def describe(self) -> pd.Series:
    """Statistics of the deviations, like `pandas.DataFrame.describe`.

    Position statistics are in meters, rotation statistics in radians; rotation statistics are NaN
    without orientations. ``pos_mean`` is the position ATE, ``rot_mean`` the rotation ATE.

    Returns:
        pd.Series: Statistics, named after the result.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> ate_result = tpy.ate(estimate, reference)
        >>> ate_result.describe()[["poses", "pos_mean", "pos_rms"]].round(3)
        poses       4541.000
        pos_mean       1.153
        pos_rms        1.304
        Name: KITTI ORB vs. KITTI Ground Truth, dtype: float64
    """
    statistics: dict[str, float] = {
        "poses": float(len(self)),
        "pos_mean": float(self.pos_ate),
        "pos_median": float(self.pos_dev_median),
        "pos_rms": float(self.pos_dev_rms),
        "pos_std": float(self.pos_dev_std),
        "pos_p95": float(self.pos_dev_p95),
        "pos_min": float(self.pos_dev_min),
        "pos_max": float(self.pos_dev_max),
        "pos_bias_x": float(self.pos_bias_x),
        "pos_bias_y": float(self.pos_bias_y),
        "pos_bias_z": float(self.pos_bias_z),
        "pos_bias_along": float(self.pos_bias_along),
        "pos_bias_cross_h": float(self.pos_bias_cross_h),
        "pos_bias_cross_v": float(self.pos_bias_cross_v),
    }
    rotation_keys = ("rot_mean", "rot_rms", "rot_std", "rot_p95", "rot_max")
    if self.has_orientation:
        values = (self.rot_ate, self.rot_dev_rms, self.rot_dev_std, self.rot_dev_p95, self.rot_dev_max)
        statistics |= {key: float(value) for key, value in zip(rotation_keys, values)}
    else:
        statistics |= dict.fromkeys(rotation_keys, np.nan)
    return pd.Series(statistics, name=self.name)

    def remove_ate_above(self, threshold: float) -> None:
    """Removes the deviations (and their poses) whose position deviation is not below a threshold.

    This is useful to remove outliers from the ATE.

    Args:
        threshold (float): Deviations of this size (m) and larger are removed; 0 or less keeps all.
    """
    if threshold <= 0.0:
        return

    ate_below_idx = self.pos_dev_comb < threshold
    self.trajectory.mask(ate_below_idx, inplace=True)

    self.abs_dev.pos_dev = self.abs_dev.pos_dev[ate_below_idx]
    self.abs_dev.directed_pos_dev = self.abs_dev.directed_pos_dev[ate_below_idx]

    if self.abs_dev.rot_dev is not None:
        # selected as quaternions, a round trip through Euler angles is inexact near gimbal lock
        self.abs_dev.rot_dev = self.abs_dev.rot_dev[ate_below_idx]

    # the combined deviations and rotation statistics are cached and depend on the removed deviations
    for cached in _CACHED:
        self.__dict__.pop(cached, None)

    def confidence_interval(
    self, *, quantity: str = "position", statistic: str = "mean", confidence: float = DEFAULT_CONFIDENCE
) -> tuple[float, float]:
    """Confidence interval of the mean (ATE) or RMS of the position or rotation deviations.

    The deviations are usually correlated in time, so that the interval is derived with a moving block
    bootstrap whose blocks are twice as long as the correlation time. It describes how precisely the
    deviations of this evaluation determine the statistic, e.g. whether two estimates differ significantly. It is
    reproducible (fixed seed) and cached.

    Args:
        quantity (str, optional): ``position`` (meters) or ``rotation`` (radians), case-insensitive.
            Defaults to ``position``.
        statistic (str, optional): ``mean`` or ``rms``, case-insensitive. Defaults to ``mean``.
        confidence (float, optional): Confidence level. Defaults to 0.95.

    Returns:
        tuple[float, float]: Lower and upper bound.

    References:
        1. H. R. Künsch (1989): The jackknife and the bootstrap for general stationary observations. The Annals of Statistics 17(3), 1217-1241.
    """
    quantity, statistic = quantity.lower(), statistic.lower()
    if quantity not in ("position", "rotation"):
        raise ValueError(f"Unknown quantity '{quantity}', use 'position' or 'rotation'.")
    if statistic not in ("mean", "rms"):
        raise ValueError(f"Unknown statistic '{statistic}', use 'mean' or 'rms'.")
    key = (quantity, statistic, confidence)
    if key not in self._confidence_intervals:
        values = self.pos_dev_comb if quantity == "position" else self.rot_dev_comb
        order = np.argsort(self.trajectory.timestamps, kind="stable")  # chronological, also if sorted spatially
        self._confidence_intervals[key] = block_bootstrap_ci(
            values[order], self.trajectory.timestamps[order], statistic=statistic, confidence=confidence
        )
    return self._confidence_intervals[key]

    def segment_statistics(
    self,
    *,
    segment_length: float | None = None,
    by: str = "time",
    labels: np.ndarray | list | None = None,
) -> pd.DataFrame:
    """Statistics of the deviations in segments of the trajectory, e.g. to compare sections of different
    conditions (tunnel and open sky) or to see whether the accuracy changes over time.

    Segments are either consecutive windows of ``segment_length`` seconds (``by="time"``) or meters
    (``by="path_length"``), or given by a label of each deviation (``labels``, e.g. from a classification of
    the environment).

    Args:
        segment_length (float | None, optional): Length of the windows. Required unless labels are given.
        by (str, optional): ``time`` or ``path_length``. Defaults to ``time``.
        labels (np.ndarray | list | None, optional): Segment label of each deviation (in the current order of
            the result). Overrides segment_length.

    Returns:
        pd.DataFrame: One row per segment with the columns ``segment``, ``start`` and ``end`` (smallest and
        largest time or path length), ``count``, ``pos_mean``, ``pos_median``, ``pos_rms``, ``pos_p95``,
        ``pos_max`` (meters) and, with orientations, ``rot_mean``, ``rot_rms`` and ``rot_max`` (degrees).
    """
    by = by.lower()
    if by not in ("time", "path_length"):
        raise ValueError(f"Unknown segment index '{by}', use 'time' or 'path_length'.")
    index = self.trajectory.timestamps if by == "time" else self.trajectory.path_lengths
    if labels is not None:
        segment = np.asarray(labels)
        if len(segment) != len(self.pos_dev_comb):
            raise ValueError(f"{len(segment)} labels were given for {len(self.pos_dev_comb)} deviations.")
    elif segment_length is not None:
        segment = segment_ids(index, segment_length)
    else:
        raise ValueError("Give either a segment length or labels.")

    frame = pd.DataFrame({"segment": segment, "index": index, "pos": self.pos_dev_comb})
    if self.has_orientation:
        frame["rot"] = np.rad2deg(self.rot_dev_comb)

    def p95(values: pd.Series) -> float:
        return float(np.percentile(values, 95))

    def rms_of(values: pd.Series) -> float:
        return float(rms(values.to_numpy()))

    grouped = frame.groupby("segment", sort=True)
    table = grouped.agg(
        start=("index", "min"),
        end=("index", "max"),
        count=("pos", "size"),
        pos_mean=("pos", "mean"),
        pos_median=("pos", "median"),
        pos_rms=("pos", rms_of),
        pos_p95=("pos", p95),
        pos_max=("pos", "max"),
    )
    if self.has_orientation:
        table = table.join(grouped.agg(rot_mean=("rot", "mean"), rot_rms=("rot", rms_of), rot_max=("rot", "max")))
    return table.reset_index()

    Container holding absolute pose deviations between two trajectories.

    The arrays stored in this dataclass represent per-epoch deviations after an optional temporal/spatial alignment of a reference (ground truth) and a test trajectory.     All deviations describe how far the evaluated (test) trajectory deviates from the reference, i.e. test minus reference: a positive x deviation means the test trajectory is further in x direction, a positive along-track deviation that it is ahead, a positive horizontal cross-track deviation that it is to the right and a positive vertical cross-track deviation that it is above.

      3D array of directed position deviation vectors (e.g. along-track, horizontal/vertical cross-track) in meters. Shape is (N, 3). These deviations are independent of the global coordinate frame, as they are derived based on the along-track direction of the trajectory.

def from_ate_result(
    cls, ate_result: "ATEResult", *, max_lag: float = DEFAULT_MAX_LAG, window: float = DEFAULT_WINDOW
) -> "DeviationCorrelation":
    """Analyzes the deviations of an ATE result in chronological order.

    Args:
        ate_result (ATEResult): Result to analyze.
        max_lag (float, optional): Largest lag of the autocorrelation [s], at most n / 10 sampling
            intervals. Defaults to 600 s.
        window (float, optional): Length of the windows of the windowed standard deviations [s], at most a
            tenth of the duration (so that there are at least 10 windows). Defaults to 60 s.

    Returns:
        DeviationCorrelation: Correlation of every deviation component and the cross-correlations.
    """
    timestamps = ate_result.trajectory.timestamps
    if (duration := float(np.ptp(timestamps)) if len(timestamps) else 0.0) > 0:
        window = min(window, duration / MIN_WINDOWS)
    values = {
        "along": ate_result.pos_dev_along,
        "cross_h": ate_result.pos_dev_cross_h,
        "cross_v": ate_result.pos_dev_cross_v,
        "x": ate_result.pos_dev_x,
        "y": ate_result.pos_dev_y,
        "z": ate_result.pos_dev_z,
    }
    if ate_result.has_orientation:
        values |= {"roll": ate_result.rot_dev_x, "pitch": ate_result.rot_dev_y, "yaw": ate_result.rot_dev_z}

    step = sampling_interval(timestamps)
    correlation = cls(sampling_interval=step, num_samples=len(timestamps))
    # all components at once: the grid of the epochs and the pairs of each lag are the same
    lags, rho, pairs = autocorrelation(timestamps, np.column_stack(list(values.values())), max_lag=max_lag)
    for column, (name, component) in enumerate(values.items()):
        # white noise (e.g. of the reference) drops at once and would hide the correlated part
        correlated = white_noise_split(lags, rho[:, column], pairs)
        tau = correlation_time(lags, rho[:, column], level=correlated) if correlated > 0 else 0.0
        window_centers, window_std = windowed_std(timestamps, component, window)
        correlation.components[name] = ComponentCorrelation(
            lags=lags,
            autocorrelation=rho[:, column],
            pairs=pairs,
            white_noise_share=1.0 - correlated,
            correlation_time=tau,
            # the time covered by the deviations, without gaps
            effective_samples=effective_samples(len(component), len(component) * step, tau),
            std=float(np.std(component)) if len(component) else float("nan"),
            window_centers=window_centers,
            window_std=window_std,
        )

    correlation.cross_correlation_directed = _correlation_matrix(ate_result.abs_dev.directed_pos_dev)
    correlation.cross_correlation_xyz = _correlation_matrix(ate_result.abs_dev.pos_dev)
    return correlation

    Class representing a stochastic parameter with variance.

    A parameter holds a stochastic value with some variance and can be enabled or disabled. When disabled, the parameter's value equals its default value. The default value should be chosen so the parameter has no effect on computations (e.g., scale=1, rotation/translation=0).

    Parameter disabling is used to exclude parameters from least-squares estimation during trajectory alignment.

    Attributes:

    • default –

      Default value used when parameter is disabled

    • enabled –

      Whether the parameter is active

    • name –

      Parameter name for identification

    • unit –

      Unit of measurement for the parameter

    Examples:

    >>> import trajectopy as tpy
>>> scale = tpy.Parameter(value=1.0001, variance=1e-8, default=1.0, name="Scale")
>>> scale.disable()
>>> scale.value  # a disabled parameter has its default value
1.0

    def __str__(self) -> str:
    """Returns formatted string representation of the parameter.

    Returns:
        str: Formatted string with parameter name, value, unit, and standard deviation
    """
    formatter_value = UNIT_FORMAT_RULES.get(self.unit, {"normal": (1, "-", 4)})["normal"]
    formatter_variance = UNIT_FORMAT_RULES.get(self.unit, {"precise": (1, "-", 2)}).get("precise", formatter_value)
    # "-" for an invalid variance, e.g. of a parameter that the data does not determine
    std = (
        f"{np.sqrt(self.variance) * formatter_variance[0]:<8.{formatter_variance[2]}f}{formatter_variance[1]:<4}"
        if self.variance >= 0
        else f"{'-':<12}"
    )
    return f"{self.name:<16} = {self.value * formatter_value[0]:<8.{formatter_value[2]}f}{formatter_value[1]:<4}  s-dev.: {std}"
