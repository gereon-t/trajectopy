    Processing¶

    Functions that align, match, resample, smooth, sort and combine trajectories. They return new trajectories and leave their inputs unchanged unless inplace=True is passed. See the user guide on alignment and trajectories.

    def estimate_alignment(
    trajectory: Trajectory,
    other: Trajectory,
    settings: AlignmentSettings | ProcessingSettings | None = None,
    *,
    initial_parameters: AlignmentParameters | None = None,
) -> AlignmentResult:
    """Estimates the alignment between two trajectories.

    Performs Helmert, Leverarm, and Time shift estimation depending on the configuration.
    This function only estimates the alignment parameters, it does not apply them to the trajectory.

    Args:
        trajectory (Trajectory): Trajectory to align.
        other (Trajectory): Reference trajectory to align to.
        settings (AlignmentSettings | ProcessingSettings, optional): Alignment settings, or processing
            settings whose alignment and matching settings are used. Defaults to the default settings.
        initial_parameters (AlignmentParameters, optional): Prior (e.g. from the interactive alignment) used as
            start values of the least squares adjustment instead of the direct initialization. Parameters
            that are not estimated stay fixed at their prior value. A time shift of the prior may be large
            (e.g. trajectories recorded with unrelated clocks): the timestamps are shifted by it before the
            trajectories are matched, and only the remaining time shift is estimated. Defaults to None.

    Returns:
        AlignmentResult: Result of the alignment process.

    See Also:
        [`trajectopy.apply_alignment`][trajectopy.apply_alignment]: applies the result.
        [`trajectopy.align`][trajectopy.align]: both steps at once.
        [`trajectopy.AlignmentSettings`][trajectopy.AlignmentSettings]: which parameters are estimated.

    References:
        1. G. Tombrink, A. Dreier, L. Klingbeil, H. Kuhlmann (2024): Spatio-temporal trajectory alignment for trajectory evaluation. Journal of Applied Geodesy. https://doi.org/10.1515/jag-2024-0040

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> alignment = tpy.estimate_alignment(estimate, reference)
        >>> alignment.converged
        True
        >>> print(f"{alignment.position_parameters.sim_trans_x.value:.3f} m")
        -1.363 m
        >>> aligned = tpy.apply_alignment(estimate, alignment)
    """
    alignment_settings, matching_settings = _alignment_and_matching_settings(settings)

    logger.info("Aligning trajectory positions ...")

    # The time shift is modelled by the velocity (x + v * dt), which only holds for small shifts. A prior time
    # shift is therefore applied to the timestamps beforehand and added to the estimated (remaining) one.
    prior_time_shift = 0.0 if initial_parameters is None else float(initial_parameters.time_shift.value)
    if initial_parameters is not None and prior_time_shift != 0.0:
        trajectory = trajectory.copy()
        trajectory.timestamps = trajectory.timestamps - prior_time_shift
        initial_parameters = copy.deepcopy(initial_parameters)
        initial_parameters.time_shift.value = 0.0
        initial_parameters.time_shift.default = 0.0

    alignment_data = AlignmentData(
        traj_from=trajectory,
        traj_to=other,
        alignment_settings=alignment_settings,
        matching_settings=matching_settings,
    )
    ghm_alignment = AlignmentEstimator(alignment_data=alignment_data, initial_parameters=initial_parameters)
    estimated_parameters = ghm_alignment.estimate_parameters()
    if prior_time_shift != 0.0:
        time_shift = estimated_parameters.time_shift
        time_shift.value = time_shift.value + prior_time_shift
        time_shift.default = time_shift.default + prior_time_shift

    pre_aligned_rotations = (
        alignment_data.traj_from.transform(estimated_parameters.sim3_matrix, inplace=False).rotations
        if alignment_settings.estimation_settings.sensor_rotation
        else None
    )
    if pre_aligned_rotations is not None and alignment_data.traj_to.rotations is not None:
        logger.info("Aligning rotations ...")
        sensor_rot_params = align_rotations(rot_from=pre_aligned_rotations, rot_to=alignment_data.traj_to.rotations)
        logger.info("Estimated sensor rotation:\n%s", sensor_rot_params)
    else:
        sensor_rot_params = SensorRotationParameters(enabled=False)

    return AlignmentResult(
        name=f"{alignment_data.traj_from.name} to {alignment_data.traj_to.name}",
        position_parameters=estimated_parameters,
        rotation_parameters=sensor_rot_params,
        estimation_of=ghm_alignment.settings.estimation_settings,
        # without least squares parameters there is nothing that could fail to converge
        converged=ghm_alignment.has_results or alignment_settings.estimation_settings.all_lq_disabled,
    )

    def apply_alignment(
    trajectory: Trajectory, alignment_result: AlignmentResult, *, inplace: bool = False
) -> "Trajectory":
    """Transforms trajectory using alignment parameters.

    After computing the alignment parameters needed to align two trajectories, they can be
    applied to arbitrary trajectories.

    Args:
        trajectory (Trajectory): Trajectory to apply alignment to.
        alignment_result (AlignmentResult): Alignment result containing transformation parameters.
        inplace (bool, optional): Perform in-place. Defaults to False.

    Returns:
        Trajectory: Aligned trajectory.

    See Also:
        [`trajectopy.estimate_alignment`][trajectopy.estimate_alignment]: estimates the alignment.
        [`trajectopy.align`][trajectopy.align]: both steps at once.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> alignment = tpy.estimate_alignment(estimate, reference)
        >>> aligned = tpy.apply_alignment(estimate, alignment)
        >>> len(aligned) == len(estimate)
        True
    """

    def _prepare_alignment_application(
        trajectory: Trajectory, alignment_parameters: AlignmentParameters
    ) -> tuple[np.ndarray, np.ndarray, np.ndarray, float, float, float]:
        if trajectory.rotations is not None:
            rpy = trajectory.rotations.as_euler("xyz", degrees=False)
            euler_x, euler_y, euler_z = rpy[:, 0], rpy[:, 1], rpy[:, 2]
            lever_x, lever_y, lever_z = (
                alignment_parameters.lever_x.value,
                alignment_parameters.lever_y.value,
                alignment_parameters.lever_z.value,
            )
        else:
            if any(
                parameter.value != 0.0
                for parameter in (
                    alignment_parameters.lever_x,
                    alignment_parameters.lever_y,
                    alignment_parameters.lever_z,
                )
            ):
                logger.warning("Trajectory has no orientations. Cannot apply leverarm.")
            euler_x = euler_y = euler_z = np.zeros(len(trajectory))
            lever_x, lever_y, lever_z = 0.0, 0.0, 0.0

        return euler_x, euler_y, euler_z, lever_x, lever_y, lever_z

    trajectory = trajectory if inplace else trajectory.copy()
    has_orientations = trajectory.has_orientation

    # leverarm and time
    (
        euler_x,
        euler_y,
        euler_z,
        lever_x,
        lever_y,
        lever_z,
    ) = _prepare_alignment_application(trajectory, alignment_result.position_parameters)

    # do not use velocities for time shift application
    trafo_x, trafo_y, trafo_z = leverarm_time_component(
        euler_x=euler_x,
        euler_y=euler_y,
        euler_z=euler_z,
        lever_x=lever_x,
        lever_y=lever_y,
        lever_z=lever_z,
        time_shift=0,
        speed_x=0,
        speed_y=0,
        speed_z=0,
    )
    trajectory.positions.xyz += np.c_[trafo_x, trafo_y, trafo_z]

    # instead, apply time shift as a simple shift of the timestamps
    trajectory.timestamps -= alignment_result.position_parameters.time_shift.value
    logger.info(
        f"Applied time shift of {alignment_result.position_parameters.time_shift.value * 1000:.1f} ms to timestamps."
    )

    # similiarity transformation
    trajectory.transform(alignment_result.position_parameters.sim3_matrix, inplace=True)

    logger.info("Applied alignment parameters to positions.")

    # sensor orientation (mounting rotation in the body frame)
    if trajectory.rotations is not None:
        trajectory.rotations = trajectory.rotations * alignment_result.rotation_parameters.rotation_set
        logger.info("Applied alignment parameters to orientations.")

    if not has_orientations:
        trajectory.rotations = None

    return trajectory

    def align(
    trajectory: Trajectory,
    other: Trajectory,
    settings: AlignmentSettings | ProcessingSettings | None = None,
    *,
    inplace: bool = False,
) -> Trajectory:
    """Aligns the trajectory with another trajectory.

    Args:
        trajectory (Trajectory): Trajectory to align.
        other (Trajectory): Reference trajectory.
        settings (AlignmentSettings | ProcessingSettings, optional): Alignment settings, or processing
            settings whose alignment and matching settings are used. Defaults to the default settings.
        inplace (bool, optional): Perform in-place. Defaults to False.

    Returns:
        Trajectory: Aligned trajectory.

    See Also:
        [`trajectopy.estimate_alignment`][trajectopy.estimate_alignment], [`trajectopy.apply_alignment`][trajectopy.apply_alignment]: the single steps.

    References:
        1. G. Tombrink, A. Dreier, L. Klingbeil, H. Kuhlmann (2024): Spatio-temporal trajectory alignment for trajectory evaluation. Journal of Applied Geodesy. https://doi.org/10.1515/jag-2024-0040

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> aligned = tpy.align(estimate, reference)
        >>> len(aligned) == len(estimate)
        True
    """
    alignment = estimate_alignment(trajectory, other, settings)
    return apply_alignment(trajectory=trajectory, alignment_result=alignment, inplace=inplace)

    match_trajectories(
    trajectory: Trajectory,

    | ProcessingSettings

    Matches two trajectories using the specified method.

    Supported methods:

    • INTERPOLATION: Interpolates the test trajectory onto the reference trajectory using its timestamps. The interpolation is linear for both positions and rotations (SLERP).
    • NEAREST_TEMPORAL: Finds the nearest temporal match without interpolation by finding the nearest timestamp in the target trajectory for each timestamp in the source trajectory.
    • NEAREST_SPATIAL: Finds the nearest spatial match without interpolation by finding the nearest pose in the target trajectory for each pose in the source trajectory using Euclidean distance.
    • NEAREST_SPATIAL_INTERPOLATED: Finds the nearest k spatial matches and spatially interpolates using a 3d line; the orientations of the reference are averaged over the same nearest poses, the trajectory keeps its own. Both trajectories will have the length of the test trajectory.

    Parameters:

    • trajectory (Trajectory) –

      Trajectory to match.

    • other (Trajectory) –

      Other trajectory to match against.

    • settings (MatchingSettings | ProcessingSettings, default: None ) –

      Matching settings, or processing settings whose matching settings are used. Defaults to the default settings.

    • inplace (bool, default: False ) –

      Whether to modify the input trajectories. Defaults to False.

    Returns:

    • tuple[Trajectory, Trajectory] –

      Tuple[Trajectory, Trajectory]: Matched trajectories.

    See Also

    trajectopy.MatchingSettings: the methods. trajectopy.interpolate: interpolation at given timestamps.

    def match_trajectories(
    trajectory: Trajectory,
    other: Trajectory,
    settings: MatchingSettings | ProcessingSettings | None = None,
    *,
    inplace: bool = False,
) -> tuple[Trajectory, Trajectory]:
    """Matches two trajectories using the specified method.

    Supported methods:

    - **INTERPOLATION**: Interpolates the test trajectory onto the reference trajectory using
      its timestamps. The interpolation is linear for both positions and rotations (SLERP).
    - **NEAREST_TEMPORAL**: Finds the nearest temporal match without interpolation by finding
      the nearest timestamp in the target trajectory for each timestamp in the source trajectory.
    - **NEAREST_SPATIAL**: Finds the nearest spatial match without interpolation by finding
      the nearest pose in the target trajectory for each pose in the source trajectory using
      Euclidean distance.
    - **NEAREST_SPATIAL_INTERPOLATED**: Finds the nearest k spatial matches and spatially
      interpolates using a 3d line; the orientations of the reference are averaged over the same nearest
      poses, the trajectory keeps its own. Both trajectories will have the length of the test trajectory.

    Args:
        trajectory (Trajectory): Trajectory to match.
        other (Trajectory): Other trajectory to match against.
        settings (MatchingSettings | ProcessingSettings, optional): Matching settings, or processing settings
            whose matching settings are used. Defaults to the default settings.
        inplace (bool, optional): Whether to modify the input trajectories. Defaults to False.

    Returns:
        Tuple[Trajectory, Trajectory]: Matched trajectories.

    See Also:
        [`trajectopy.MatchingSettings`][trajectopy.MatchingSettings]: the methods.
        [`trajectopy.interpolate`][trajectopy.interpolate]: interpolation at given timestamps.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> estimate = tpy.Trajectory.from_file("./example_data/KITTI_ORB.traj")
        >>> matched, matched_reference = tpy.match_trajectories(
        ...     estimate, reference, tpy.MatchingSettings(method="nearest_temporal")
        ... )
        >>> len(matched) == len(matched_reference)
        True
    """
    matching_settings = resolve_settings(settings, MatchingSettings)

    trajectory = trajectory if inplace else trajectory.copy()
    other = other if inplace else other.copy()

    if (
        trajectory.sorting == Sorting.PATH_LENGTH
        and other.sorting == Sorting.PATH_LENGTH
        and matching_settings.method in [MatchingMethod.NEAREST_TEMPORAL, MatchingMethod.INTERPOLATION]
        and not np.array_equal(trajectory.index, other.index)
    ):
        _adopt_reference_path_lengths(trajectory=trajectory, other=other)

    if not trajectory.overlaps_with(other) and matching_settings.method in [
        MatchingMethod.NEAREST_TEMPORAL,
        MatchingMethod.INTERPOLATION,
    ]:
        logger.warning("Trajectories do not overlap! Performing rough matching first.")
        timeshift = match_non_overlapping_stamps(
            trajectory=other, other=trajectory, max_distance=matching_settings.max_distance
        )
        logger.info("Rough matching index offset: %.3f", timeshift)

        if trajectory.sorting == Sorting.TIME:
            trajectory.timestamps = trajectory.timestamps + timeshift
        else:
            trajectory.path_lengths = trajectory.path_lengths + timeshift

    logger.info("Matching trajectories using method %s", matching_settings.method.name)

    if matching_settings.method == MatchingMethod.INTERPOLATION:
        return _match_trajectories_interpolation(
            trajectory=trajectory, other=other, max_gap_size=matching_settings.max_gap_size
        )

    if matching_settings.method == MatchingMethod.NEAREST_TEMPORAL:
        return _match_trajectories_temporal(
            trajectory=trajectory, other=other, max_distance=matching_settings.max_time_diff
        )

    if matching_settings.method == MatchingMethod.NEAREST_SPATIAL:
        return _match_trajectories_spatial(
            trajectory=trajectory, other=other, max_distance=matching_settings.max_distance
        )

    if matching_settings.method == MatchingMethod.NEAREST_SPATIAL_INTERPOLATED:
        return _match_trajectories_spatial_interpolation(
            trajectory=trajectory,
            other=other,
            max_distance=matching_settings.max_distance,
            k_nearest=matching_settings.k_nearest,
        )

    raise ValueError(f"Matching method {matching_settings.method} not supported!")

    def approximate(
    trajectory: Trajectory,
    settings: ApproximationSettings | ProcessingSettings | None = None,
    *,
    inplace: bool = False,
) -> Trajectory:
    """Approximate the trajectory using piecewise cubic polynomials based on the provided settings.

    Args:
        trajectory (Trajectory): The trajectory to approximate.
        settings (ApproximationSettings | ProcessingSettings, optional): Approximation settings, or processing
            settings whose approximation settings are used. Defaults to the default settings.
        inplace (bool): If True, modify the trajectory in place. Defaults to False.

    Returns:
        Trajectory: The approximated trajectory.

    See Also:
        [`trajectopy.ApproximationSettings`][trajectopy.ApproximationSettings]: the options.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> smoothed = tpy.approximate(reference, tpy.ApproximationSettings(position_interval_size=50.0))
        >>> len(smoothed) == len(reference)
        True
    """
    approximation_settings = resolve_settings(settings, ApproximationSettings)

    if approximation_settings.position_approximation_method == PositionApproximationMethod.CUBIC:
        xyz_approx = _piecewise_cubic(
            index=trajectory.index,
            values=trajectory.xyz,
            min_win_size=approximation_settings.position_interval_size,
            min_obs=approximation_settings.position_min_observations,
        )
    else:
        raise ValueError(
            f"Unknown position approximation method: {approximation_settings.position_approximation_method}"
        )

    traj_approx = trajectory if inplace else trajectory.copy()
    traj_approx.positions.xyz = xyz_approx

    if not traj_approx.has_orientation:
        return traj_approx

    if approximation_settings.rotation_approximation_method == RotationApproximationMethod.AVERAGE_PER_PATH_LENGTH:
        quat_approx = _average_rotations_per_path_length(
            trajectory, max_gap_size=approximation_settings.max_path_length_gap
        )
    elif approximation_settings.rotation_approximation_method == RotationApproximationMethod.AVERAGE_IN_WINDOW:
        quat_approx = _average_rotations_in_window(
            index=trajectory.index,
            quat=trajectory.quat,
            win_size=approximation_settings.rotation_window_size,
        )
    else:
        raise ValueError(
            f"Unknown rotation approximation method: {approximation_settings.rotation_approximation_method}"
        )
    traj_approx.rotations = Rotation.from_quat(quat_approx)

    return traj_approx

    def divide_into_laps(
    trajectory: Trajectory, *, return_lap_indices: bool = False
) -> list[Trajectory] | tuple[list[Trajectory], np.ndarray]:
    """Divides a spatially sorted trajectory into laps.

    This is only useful for trajectories that describe a closed loop without intersections. A new lap starts
    where the path length of the spatially sorted trajectory jumps back to the start of the loop.

    Args:
        trajectory (Trajectory): Trajectory sorted by path length (see [`sort_spatially`][trajectopy.sort_spatially]).
        return_lap_indices (bool, optional): Whether to return lap indices. Defaults to False.

    Returns:
        list[Trajectory] | tuple[list[Trajectory], np.ndarray]: The laps, each sorted by path length. With
        ``return_lap_indices=True``, a tuple of the laps and the indices of the chronologically ordered
        trajectory at which the laps start (followed by the number of poses).

    Raises:
        ValueError: If the trajectory is not sorted by path length.

    See Also:
        [`trajectopy.sort_spatially`][trajectopy.sort_spatially]: sorts a trajectory spatially first.

    Examples:
        >>> import numpy as np
        >>> import trajectopy as tpy
        >>> angle = np.linspace(0.0, 4 * np.pi, 400, endpoint=False)  # two laps
        >>> loop = tpy.sort_spatially(tpy.Trajectory(np.column_stack([np.cos(angle), np.sin(angle), 0 * angle])))
        >>> len(tpy.divide_into_laps(loop))
        2
    """
    if trajectory.sorting != Sorting.PATH_LENGTH:
        logger.error("Provide a trajectory sorted by path length to divide into laps.")
        raise ValueError("Trajectory must be sorted by path length to divide into laps.")
    else:
        trajectory = trajectory.copy()

    trajectory.set_sorting(Sorting.TIME)
    arc_length_diffs = np.diff(trajectory.path_lengths)
    arc_length_threshold = 0.95 * np.max(trajectory.path_lengths)
    lap_indices = np.r_[0, np.where(np.abs(arc_length_diffs) > arc_length_threshold)[0], len(trajectory)]

    # divide into laps
    laps = []
    for i in range(len(lap_indices) - 1):
        lap = trajectory.mask(np.arange(lap_indices[i], lap_indices[i + 1]), inplace=False)
        lap.set_sorting(Sorting.PATH_LENGTH)
        laps.append(lap)

    return laps if (not return_lap_indices) else (laps, lap_indices)

    def merge_trajectories(trajectories: Iterable[Trajectory]) -> Trajectory:
    """Merges a list of trajectories into one trajectory.

    This function ignores EPSG codes and merges the trajectories based on their timestamps.
    Therefore, all trajectories should be in the same coordinate system before merging.

    Args:
        trajectories (Iterable[Trajectory]): List of trajectories to merge.

    Returns:
        Trajectory: Merged trajectory.

    Raises:
        ValueError: If no trajectories are provided.

    See Also:
        [`trajectopy.average_trajectories`][trajectopy.average_trajectories]: pose-wise mean instead.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> merged = tpy.merge_trajectories([reference.crop(0.0, 9.0), reference.crop(10.0, 19.0)])
        >>> len(merged)
        20
    """
    trajectories = list(trajectories)
    if not trajectories:
        raise ValueError("Cannot merge empty list of trajectories")

    epsg_set = {t.positions.epsg for t in trajectories}

    if len(epsg_set) > 1:
        logger.warning(
            "Merging trajectories with different EPSG codes. "
            "This may lead to unexpected results. "
            "Consider reprojecting the trajectories to the same EPSG code."
        )

    epsg = trajectories[0].positions.epsg  # the first trajectory's, not an arbitrary one of the set

    merged_xyz = np.concatenate([t.positions.xyz for t in trajectories], axis=0)
    # orientations only if all trajectories have them: any substitute would be made-up data
    with_orientation = all(t.rotations is not None for t in trajectories)
    if not with_orientation and any(t.rotations is not None for t in trajectories):
        logger.warning("Not all merged trajectories have orientations. The merged trajectory has none.")
    merged_rotations = (
        Rotation.from_quat(np.concatenate([t.quat for t in trajectories], axis=0)) if with_orientation else None
    )
    merged_timestamps = np.concatenate([t.timestamps for t in trajectories], axis=0)
    merged_velocities = np.concatenate([t.velocity_xyz for t in trajectories], axis=0)
    merged_std = (
        np.concatenate([t.position_std for t in trajectories if t.position_std is not None], axis=0)
        if all(t.position_std is not None for t in trajectories)
        else None
    )

    # the trajectory orders the poses chronologically before it derives the path lengths
    return Trajectory(
        name="Merged",
        timestamps=merged_timestamps,
        positions=Positions(xyz=merged_xyz, epsg=epsg),
        rotations=merged_rotations,
        velocity_xyz=merged_velocities,
        position_std=merged_std,
    )

    def average_trajectories(trajectories: Iterable[Trajectory]) -> Trajectory:
    """Averages a list of trajectories into one trajectory.

    This function ignores EPSG codes and averages trajectories.

    Args:
        trajectories (Iterable[Trajectory]): List of trajectories to average.

    Returns:
        Trajectory: Averaged trajectory.

    Raises:
        ValueError: If trajectories have different lengths or no trajectories provided.

    See Also:
        [`trajectopy.merge_trajectories`][trajectopy.merge_trajectories]: concatenates trajectories instead.

    Examples:
        >>> import trajectopy as tpy
        >>> reference = tpy.Trajectory.from_file("./example_data/KITTI_gt.traj")
        >>> averaged = tpy.average_trajectories([reference, reference.copy()])
        >>> import numpy as np
        >>> np.allclose(averaged.xyz, reference.xyz)
        True
    """
    trajectories = list(trajectories)
    if not trajectories:
        raise ValueError("Cannot average empty list of trajectories")

    lengths = {len(t) for t in trajectories}

    if len(lengths) != 1:
        raise ValueError(
            "Trajectories must have the same length to be averaged. Ensure this using trajectory matching."
        )

    xyz_stack = np.stack([t.positions.xyz for t in trajectories], axis=0)

    rots = [t.rotations.as_quat() for t in trajectories if t.rotations is not None]
    mean_rots = []
    for i in range(len(trajectories[0])):
        rotations = [rot[i] for rot in rots]
        if not rotations:
            continue

        mean_rots.append(Rotation.from_quat(np.array(rotations)).mean().as_quat())

    avg_xyz = np.mean(xyz_stack, axis=0)

    return Trajectory(
        name="Averaged",
        timestamps=trajectories[0].timestamps,
        positions=Positions(xyz=avg_xyz, epsg=trajectories[0].positions.epsg),
        rotations=Rotation.from_quat(np.array(mean_rots)) if mean_rots else None,
    )
